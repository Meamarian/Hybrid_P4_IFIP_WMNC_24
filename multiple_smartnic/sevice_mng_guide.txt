Environment=DETECT_MAC=yes                                                                                                                                                                                                       
Environment=LD_LIBRARY_PATH=/opt/netronome/lib:/opt/nfp_pif/lib:$LD_LIBRARY_PATH

[Install]                                                                                                                                                                                                                        
WantedBy=multi-user.target
